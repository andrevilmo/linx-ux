namespace #nameSpace#
{
    public class #moduleClass# : IModule
    {
        private readonly IRegionManager regionManager;
	    private readonly #moduleClass#View defaultView;

        public #moduleClass#(IRegionManager regionManager, #moduleClass#View defaultView)
        {
            this.regionManager = regionManager;
            this.defaultView = defaultView;
        }
                
        public void Initialize()
        {
			//Initialize View Structure
            this.defaultView.ModuleName = "#moduleName#";
            this.defaultView.InitializeDataStructure();

            //Load Module
            if (this.defaultView.ModuleName != String.Empty && this.regionManager.Regions.Where(e => e.Name.Right(this.defaultView.ModuleName).IsNumeric()).Count() > 0)
            {
                string regionName = this.regionManager.Regions.Where(e => e.Name.Right(this.defaultView.ModuleName).IsNumeric()).Select(e => e.Name).Max();
                this.regionManager.AddToRegion(regionName, this.defaultView);
            }
        }        
    }
